;; QImode (8-bit) moves.
(define_expand "movqi"
  [(set (match_operand:QI 0 "nonimmediate_operand")
	(match_operand:QI 1 "general_operand"))]
  ""
  "")

(define_insn "*movqi"
  [(set (match_operand:QI 0 "nonimmediate_operand" "=q,q,m,q,m")
	(match_operand:QI 1 "general_operand"       "q,m,q,i,i"))]
  ""
  "movb\t%1, %0"
  [(set_attr "type" "move")
   (set_attr "length" "2")])

;; HImode (16-bit) moves.
(define_expand "movhi"
  [(set (match_operand:HI 0 "nonimmediate_operand")
	(match_operand:HI 1 "general_operand"))]
  ""
{
  /* On 8086, there is no push-immediate instruction.  If we are
     storing an immediate to a pre_dec (push), force it into a
     register first so the register allocator handles it properly.  */
  if (TARGET_8086
      && MEM_P (operands[0])
      && GET_CODE (XEXP (operands[0], 0)) == PRE_DEC
      && CONSTANT_P (operands[1]))
    {
      operands[1] = force_reg (HImode, operands[1]);
    }
})

(define_insn "*movhi"
  [(set (match_operand:HI 0 "nonimmediate_operand" "=r,r,m,r,m")
	(match_operand:HI 1 "general_operand"       "r,m,r,i,i"))]
  ""
  "movw\t%1, %0"
  [(set_attr "type" "move")
   (set_attr "length" "3")])

;; SImode (32-bit) moves - split into two 16-bit moves.
(define_expand "movsi"
  [(set (match_operand:SI 0 "nonimmediate_operand")
	(match_operand:SI 1 "general_operand"))]
  ""
  "")

(define_insn_and_split "*movsi"
  [(set (match_operand:SI 0 "nonimmediate_operand" "=r,r,m,r")
	(match_operand:SI 1 "general_operand"       "r,m,r,i"))]
  ""
  "#"
  "reload_completed"
  [(set (match_dup 2) (match_dup 3))
   (set (match_dup 4) (match_dup 5))]
{
  ia16_split_movsi (operands, operands + 2);
}
  [(set_attr "type" "move")])

  [(set (match_operand:SI 0 "register_operand" "=r")

		 (match_operand:SI 2 "general_operand" "rmi")))
   (clobber (reg:CC CC_REG))]

  [(parallel [(set (match_dup 3)
		   (plus:HI (match_dup 4) (match_dup 5)))
	      (clobber (reg:CC CC_REG))])
   (parallel [(set (match_dup 6)
		   (plus:HI (match_dup 7)
			    (plus:HI (match_dup 8)
				     (ltu:HI (reg:CC CC_REG) (const_int 0)))))
	      (clobber (reg:CC CC_REG))])]

  operands[5] = gen_lowpart (HImode, operands[2]);
  operands[6] = gen_highpart (HImode, operands[0]);
  operands[7] = gen_highpart (HImode, operands[1]);
  operands[8] = gen_highpart (HImode, operands[2]);

;; -------------------------------------------------------------------------
;; Shift and Rotate Instructions
;; -------------------------------------------------------------------------

;; On 8086, only shift by 1 or by CL.
;; On 80186+, immediate shift counts are supported.

;; Arithmetic shift left (QI).
(define_insn "ashlqi3"
  [(set (match_operand:QI 0 "nonimmediate_operand" "=q,q")
	(ashift:QI (match_operand:QI 1 "nonimmediate_operand" "0,0")
		   (match_operand:QI 2 "nonmemory_operand" "c,I")))
   (clobber (reg:CC CC_REG))]
  ""
{
  if (CONST_INT_P (operands[2]) && INTVAL (operands[2]) == 1)
    return "shlb\t$1, %0";
  if (CONST_INT_P (operands[2]) && TARGET_8086)
    {
      /* 8086: need to emit multiple shift-by-1.  */
      static char buf[128];
      int count = INTVAL (operands[2]);
      buf[0] = '\0';
      for (int i = 0; i < count; i++)
	strcat (buf, "shlb\t$1, %0\n\t");
      /* Remove trailing \n\t.  */
      buf[strlen (buf) - 2] = '\0';
      return buf;
    }
  return "shlb\t%2, %0";
}
  [(set_attr "type" "shift")])

(define_insn "*branch"
  [(set (pc)
	(if_then_else (match_operator 0 "comparison_operator"
		       [(reg:CC CC_REG) (const_int 0)])
		      (label_ref (match_operand 1 ""))
		      (pc)))]
  ""
{
  switch (GET_CODE (operands[0]))
    {
    case EQ:  return "je\t%l1";
    case NE:  return "jne\t%l1";
    case LT:  return "jl\t%l1";
    case GT:  return "jg\t%l1";
    case LE:  return "jle\t%l1";
    case GE:  return "jge\t%l1";
    case LTU: return "jb\t%l1";
    case GTU: return "ja\t%l1";
    case LEU: return "jbe\t%l1";
    case GEU: return "jae\t%l1";
    default:  gcc_unreachable ();
    }
}
  [(set_attr "type" "branch")
   (set_attr "length" "4")])

;; Zero-extend QI to HI.
(define_insn "zero_extendqihi2"
  [(set (match_operand:HI 0 "register_operand" "=r")
	(zero_extend:HI (match_operand:QI 1 "nonimmediate_operand" "qm")))]
  ""
{
  if (TARGET_80286 || TARGET_80186)
    return "movzbw\t%1, %0";
  /* On 8086, we need to clear the high byte manually.  */
  if (REGNO (operands[0]) == AX_REG)
    return "movb\t%1, %l0\;xorb\t%h0, %h0";
  return "movb\t%1, %l0\;xorb\t%h0, %h0";
}
  [(set_attr "type" "move")
   (set_attr "length" "4")])
