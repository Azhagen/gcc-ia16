;; QImode (8-bit) moves.
(define_expand "movqi"
  [(set (match_operand:QI 0 "nonimmediate_operand")
	(match_operand:QI 1 "general_operand"))]
  ""
{
  /* Far memory load: *far_ptr → reg.
     Split the SImode address into segment (high) and offset (low),
     load segment into ES, then use far_load_qi.  */
  if (MEM_P (operands[1]) && MEM_ADDR_SPACE (operands[1]) == ADDR_SPACE_FAR)
    {
      rtx addr = force_reg (SImode, XEXP (operands[1], 0));
      rtx seg = gen_reg_rtx (HImode);
      rtx off = gen_reg_rtx (HImode);
      emit_move_insn (off, gen_lowpart (HImode, addr));
      emit_move_insn (seg, gen_highpart (HImode, addr));
      emit_insn (gen_load_es (seg));
      emit_insn (gen_far_load_qi (operands[0], off));
      DONE;
    }
  /* Far memory store: reg → *far_ptr.  */
  if (MEM_P (operands[0]) && MEM_ADDR_SPACE (operands[0]) == ADDR_SPACE_FAR)
    {
      rtx addr = force_reg (SImode, XEXP (operands[0], 0));
      rtx seg = gen_reg_rtx (HImode);
      rtx off = gen_reg_rtx (HImode);
      rtx val = force_reg (QImode, operands[1]);
      emit_move_insn (off, gen_lowpart (HImode, addr));
      emit_move_insn (seg, gen_highpart (HImode, addr));
      emit_insn (gen_load_es (seg));
      emit_insn (gen_far_store_qi (off, val));
      DONE;
    }
  /* No mem-to-mem moves on x86.  */
  if (MEM_P (operands[0]) && MEM_P (operands[1]))
    operands[1] = force_reg (QImode, operands[1]);
})

;; HImode (16-bit) moves.
(define_expand "movhi"
  [(set (match_operand:HI 0 "nonimmediate_operand")
	(match_operand:HI 1 "general_operand"))]
  ""
{
  /* Far memory load: *far_ptr → reg.  */
  if (MEM_P (operands[1]) && MEM_ADDR_SPACE (operands[1]) == ADDR_SPACE_FAR)
    {
      rtx addr = force_reg (SImode, XEXP (operands[1], 0));
      rtx seg = gen_reg_rtx (HImode);
      rtx off = gen_reg_rtx (HImode);
      emit_move_insn (off, gen_lowpart (HImode, addr));
      emit_move_insn (seg, gen_highpart (HImode, addr));
      emit_insn (gen_load_es (seg));
      emit_insn (gen_far_load_hi (operands[0], off));
      DONE;
    }
  /* Far memory store: reg → *far_ptr.  */
  if (MEM_P (operands[0]) && MEM_ADDR_SPACE (operands[0]) == ADDR_SPACE_FAR)
    {
      rtx addr = force_reg (SImode, XEXP (operands[0], 0));
      rtx seg = gen_reg_rtx (HImode);
      rtx off = gen_reg_rtx (HImode);
      rtx val = force_reg (HImode, operands[1]);
      emit_move_insn (off, gen_lowpart (HImode, addr));
      emit_move_insn (seg, gen_highpart (HImode, addr));
      emit_insn (gen_load_es (seg));
      emit_insn (gen_far_store_hi (off, val));
      DONE;
    }
  /* No mem-to-mem moves on x86; force source through a register.  */
  if (MEM_P (operands[0]) && MEM_P (operands[1]))
    operands[1] = force_reg (HImode, operands[1]);

  /* On 8086, there is no push-immediate instruction.  If we are
     storing an immediate to a pre_dec (push), force it into a
     register first so the register allocator handles it properly.  */
  if (TARGET_8086
      && MEM_P (operands[0])
      && GET_CODE (XEXP (operands[0], 0)) == PRE_DEC
      && CONSTANT_P (operands[1]))
    operands[1] = force_reg (HImode, operands[1]);
})

  ""
  "movw\t%1, %0"
  [(set_attr "type" "move")
   (set_attr "length" "3")])

;; SImode (32-bit) push: split into two HImode pushes (high word first).
;; On 8086, immediates must go through a register since there is no
;; push-immediate instruction.
(define_insn_and_split "pushsi"
  [(set (mem:SI (pre_dec:HI (reg:HI SP_REG)))
	(match_operand:SI 0 "general_operand" "r,m,i"))]
  ""
  "#"
  ""
  [(const_int 0)]
{
  rtx lo = ia16_split_si_half (operands[0], 0);
  rtx hi = ia16_split_si_half (operands[0], 1);

  /* On 8086, force immediates into registers before pushing.  */
  if (TARGET_8086 && CONSTANT_P (hi))
    hi = force_reg (HImode, hi);
  if (TARGET_8086 && CONSTANT_P (lo))
    lo = force_reg (HImode, lo);

  emit_insn (gen_push (hi));
  emit_insn (gen_push (lo));
  DONE;
}
  [(set_attr "type" "push")])

;; Zero-extend QI to HI.
;; Use a byte move into the low half followed by an immediate move into the
;; high half.  This stays legal on 8086/80186/80286 and does not clobber
;; condition codes behind the RTL's back.
(define_insn "zero_extendqihi2"
  [(set (match_operand:HI 0 "register_operand" "=q")
	(zero_extend:HI (match_operand:QI 1 "nonimmediate_operand" "qm")))]
  ""
  "movb\t%1, %L0\;movb\t$0, %H0"
  [(set_attr "type" "move")
   (set_attr "length" "4")])

;; Same zero-extension sequence, but with flags set to match an AND mask.
(define_insn "*zero_extendqihi2_cc"

	     (match_operand:QI 1 "nonimmediate_operand" "qm")))
      (clobber (reg:CC CC_REG))])]
  ""
  "movb\t%1, %L0\;movb\t$0, %H0\;testw\t%0, %0"

(define_expand "extendhisi2"
  [(parallel [(set (match_operand:SI 0 "register_operand")
		   (sign_extend:SI (match_operand:HI 1 "register_operand")))
	      (clobber (reg:CC CC_REG))])]

  "")

;; Split into: copy low to high, then arithmetic shift right by 15
;; to fill the high word with the sign bit.
(define_insn_and_split "*extendhisi2"
  [(set (match_operand:SI 0 "register_operand" "=r")
	(sign_extend:SI (match_operand:HI 1 "register_operand" "0")))
   (clobber (reg:CC CC_REG))]
  ""
  "#"
  "reload_completed"
  [(const_int 0)]
{
  rtx lo = ia16_split_si_half (operands[0], 0);
  rtx hi = ia16_split_si_half (operands[0], 1);

  /* When allocated to AX:DX, use CWD (sign-extend AX → DX).  */
  if (REGNO (lo) == AX_REG && REGNO (hi) == DX_REG)
    {
      emit_insn (gen_rtx_SET (hi,
		   gen_rtx_ASHIFTRT (HImode,
		     gen_rtx_REG (HImode, AX_REG), GEN_INT (15))));
    }
  else
    {
      /* General case: copy low -> high, then fill high with the sign bit.
   On 8086, immediate shifts greater than 1 are not encodable here, so
   use ADD/SBB to materialize 0xffff or 0x0000 from the copied sign bit
   instead of emitting fifteen single-bit shifts.  */
      emit_insn (gen_rtx_SET (hi, lo));
      if (!TARGET_8086)
	{
	  emit_insn (gen_rtx_PARALLEL (VOIDmode, gen_rtvec (2,
	    gen_rtx_SET (hi,
	      gen_rtx_ASHIFTRT (HImode, hi, GEN_INT (15))),
	    gen_rtx_CLOBBER (VOIDmode,
	      gen_rtx_REG (CCmode, CC_REG)))));
	}
      else
	{
    /* ADD propagates the sign bit into CF, then SBB turns CF into a
       full-word sign mask.  */
    emit_insn (gen_addhi3 (hi, hi, hi));
    emit_insn (gen_rtx_PARALLEL (VOIDmode, gen_rtvec (2,
      gen_rtx_SET (hi,
        gen_rtx_MINUS (HImode, hi,
    gen_rtx_PLUS (HImode, hi,
      gen_rtx_LTU (HImode,
        gen_rtx_REG (CCmode, CC_REG), const0_rtx)))),
      gen_rtx_CLOBBER (VOIDmode,
        gen_rtx_REG (CCmode, CC_REG)))));
	}
    }
  DONE;
})

;; Zero-extend HI to SI.
;; Zero-extend HI to SI: low word = source, high word = 0.
(define_insn_and_split "zero_extendhisi2"
  [(set (match_operand:SI 0 "register_operand" "=r")
	(zero_extend:SI (match_operand:HI 1 "general_operand" "rmi")))]
  ""
  "#"
  "reload_completed"
  [(set (match_dup 2) (match_dup 1))
   (set (match_dup 3) (const_int 0))]
{

  operands[3] = ia16_split_si_half (operands[0], 1);
})
