;; HImode (16-bit) moves.
(define_expand "movhi"
  [(set (match_operand:HI 0 "nonimmediate_operand")
	(match_operand:HI 1 "general_operand"))]
  ""
{
  /* No mem-to-mem moves on x86; force source through a register.  */
  if (MEM_P (operands[0]) && MEM_P (operands[1]))
    operands[1] = force_reg (HImode, operands[1]);

  /* On 8086, there is no push-immediate instruction.  If we are
     storing an immediate to a pre_dec (push), force it into a
     register first so the register allocator handles it properly.  */
  if (TARGET_8086
      && MEM_P (operands[0])
      && GET_CODE (XEXP (operands[0], 0)) == PRE_DEC
      && CONSTANT_P (operands[1]))
    operands[1] = force_reg (HImode, operands[1]);
})

(define_insn "*movhi"
  [(set (match_operand:HI 0 "nonimmediate_operand" "=r,r,m,r,m")
	(match_operand:HI 1 "general_operand"       "r,m,r,i,i"))]
  ""
  "movw\t%1, %0"
  [(set_attr "type" "move")
   (set_attr "length" "3")])

;; SCmode (_Complex float, 64-bit) also lives in four 16-bit words.
(define_expand "movsc"
  [(set (match_operand:SC 0 "ia16_sc_nonimmediate_operand")
	(match_operand:SC 1 "ia16_sc_general_operand"))]
  ""
{
  if (push_operand (operands[0], SCmode))
    {
      ia16_emit_multiword_push (operands[1], SCmode);
      DONE;
    }

  if (GET_CODE (operands[0]) == CONCAT || GET_CODE (operands[1]) == CONCAT)
    {
      emit_move_insn (ia16_complex_part (operands[0], SCmode, 0),
		      ia16_complex_part (operands[1], SCmode, 0));
      emit_move_insn (ia16_complex_part (operands[0], SCmode, 1),
		      ia16_complex_part (operands[1], SCmode, 1));

    operands[1] = force_reg (SCmode, operands[1]);
})

;; SImode (32-bit) push: split into two HImode pushes (high word first).
;; On 8086, immediates must go through a register since there is no
;; push-immediate instruction.
(define_insn_and_split "pushsi"
  [(set (mem:SI (pre_dec:HI (reg:HI SP_REG)))
	(match_operand:SI 0 "general_operand" "r,m,i"))]
  ""
  "#"
  ""
  [(const_int 0)]
{
  rtx lo = ia16_split_si_half (operands[0], 0);
  rtx hi = ia16_split_si_half (operands[0], 1);

  /* On 8086, force immediates into registers before pushing.  */
  if (TARGET_8086 && CONSTANT_P (hi))
    hi = force_reg (HImode, hi);
  if (TARGET_8086 && CONSTANT_P (lo))
    lo = force_reg (HImode, lo);

  emit_insn (gen_push (hi));
  emit_insn (gen_push (lo));
  DONE;
}
  [(set_attr "type" "push")])

;; Named push expander for HImode.  GCC's emit_single_push_insn
;; looks up push_optab ("pushhi1") to push arguments.
;; On 8086, push-immediate is not available, so we force
;; immediates into a register first.
(define_expand "pushhi1"
  [(set (mem:HI (pre_dec:HI (reg:HI SP_REG)))
	(match_operand:HI 0 "general_operand"))]
  ""
{
  if (TARGET_8086 && CONSTANT_P (operands[0]))
    operands[0] = force_reg (HImode, operands[0]);
})

;; Named push expander for QImode.  On 8086, force immediates
;; into a register since push-immediate is not available.
;; PUSH_ROUNDING will round the 1-byte push to 2 bytes.
(define_expand "pushqi1"
  [(set (mem:QI (pre_modify:HI (reg:HI SP_REG)
		  (plus:HI (reg:HI SP_REG) (const_int -2))))
	(match_operand:QI 0 "general_operand"))]
  ""
{
  if (TARGET_8086 && CONSTANT_P (operands[0]))
    operands[0] = force_reg (QImode, operands[0]);
})

;; QI push via pre_modify (generated when PUSH_ROUNDING rounds 1 byte
;; to 2 bytes).  The 8086 always pushes 16 bits; we use pushw with
;; the word register name even though the source is QImode.

;; QI push via pre_modify: register or memory (all CPUs).
(define_insn "pushqi_pre_modify"
  [(set (mem:QI (pre_modify:HI (reg:HI SP_REG)

  return "pushw\t%0";
}
  [(set_attr "type" "push")
   (set_attr "length" "1,3")])

;; QI addition.  Use incb/decb for +1/-1 (shorter encoding).
(define_insn "addqi3"
  [(set (match_operand:QI 0 "nonimmediate_operand" "=q,q,m")
	(plus:QI (match_operand:QI 1 "nonimmediate_operand" "%0,0,0")
		 (match_operand:QI 2 "general_operand" "qi,m,qi")))
   (clobber (reg:CC CC_REG))]
  ""
{
  if (CONST_INT_P (operands[2]))
    {
      HOST_WIDE_INT val = INTVAL (operands[2]) & 0xFF;
      if (val == 1)
	return "incb\t%0";
      if (val == 0xFF)
	return "decb\t%0";
    }
  return "addb\t%2, %0";
}
  [(set_attr "type" "alu")
   (set_attr "length" "3")])

;; HI addition.  Use incw/decw for +1/-1 (shorter encoding).

		 (match_operand:HI 2 "general_operand" "ri,m,ri")))
   (clobber (reg:CC CC_REG))]
  ""
{
  if (CONST_INT_P (operands[2]))

      if (val == 1)
	return "incw\t%0";
      if (val == 0xFFFF)
	return "decw\t%0";

  return "addw\t%2, %0";
}
  [(set_attr "type" "alu")
   (set_attr "length" "4")])

      buf[strlen (buf) - 2] = '\0';

   (set (match_operand:HI 2 "register_operand")
	(match_dup 0))]
  "REGNO (operands[1]) == REGNO (operands[2])
   && peep2_reg_dead_p (2, operands[2])"
  [(set (match_dup 0) (match_dup 1))])

  [(set (match_operand:QI 0 "memory_operand")
	(match_operand:QI 1 "register_operand"))
   (set (match_operand:QI 2 "register_operand")
	(match_dup 0))]

   && peep2_reg_dead_p (2, operands[2])"
  [(set (match_dup 0) (match_dup 1))])

;; mov %reg, MEM ; mov MEM, %same_reg  ->  mov %reg, MEM
(define_peephole2
  [(set (match_operand:HI 0 "memory_operand")
	(match_operand:HI 1 "register_operand"))
   (set (match_operand:HI 2 "register_operand")
	(match_dup 0))]
  "REGNO (operands[1]) == REGNO (operands[2])"
  [(set (match_dup 0) (match_dup 1))])
